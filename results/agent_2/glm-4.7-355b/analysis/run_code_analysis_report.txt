import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import accuracy_score

df = pd.read_csv('data/iris-data.csv')
df = df.dropna(subset=['petal_width_cm'])
df['class'].replace(["Iris-setossa","versicolor"], ["Iris-setosa","Iris-versicolor"], inplace=True)
final_df = df[df['class'] != 'Iris-virginica']
final_df.loc[final_df.sepal_length_cm < 1, ['sepal_length_cm']] = final_df['sepal_length_cm']*100
final_df = final_df.drop(final_df[(final_df['class'] == "Iris-setosa") & (final_df['sepal_width_cm'] < 2.5)].index)
final_df['class'].replace(["Iris-setosa","Iris-versicolor"], [1,0], inplace=True)

inp_df = final_df.drop(final_df.columns[[4]], axis=1)
out_df = final_df.drop(final_df.columns[[0,1,2,3]], axis=1)
scaler = StandardScaler()
inp_df = scaler.fit_transform(inp_df)
X_train, X_test, y_train, y_test = train_test_split(inp_df, out_df, test_size=0.2, random_state=42)

X_tr_arr = X_train
X_ts_arr = X_test
y_tr_arr = y_train.values
y_ts_arr = y_test.values

print("X_tr_arr shape:", X_tr_arr.shape)
print("y_tr_arr shape:", y_tr_arr.shape)

# Initialize
def initialize_params(dim):
    w = np.zeros((dim, 1))
    b = 0
    return w, b

def sigmoid(z):
    return 1 / (1 + np.exp(-z))

def sigmoid_activation(result):
    final_result = 1/(1+np.exp(-result))
    return final_result

def model_optimize(w, b, X, Y):
    m = X.shape[0]
    final_result = sigmoid(np.dot(X, w) + b)
    dw = (1/m) * np.dot(X.T, (final_result - Y))
    db = (1/m) * np.sum(final_result - Y)
    grads = {"dw": dw, "db": db}
    cost = (-1/m) * np.sum(Y * np.log(final_result) + (1 - Y) * np.log(1 - final_result))
    return grads, cost

def model_predict(w, b, X, Y, learning_rate, no_iterations):
    for i in range(no_iterations):
        grads, cost = model_optimize(w, b, X, Y)
        dw = grads["dw"]
        db = grads["db"]
        w = w - (learning_rate * dw.T)
        b = b - (learning_rate * db)
    coeff = {"w": w, "b": b}
    return coeff

def predict(final_pred, m):
    y_pred = np.zeros((1,m))
    for i in range(final_pred.shape[1]):
        if final_pred[0][i] > 0.5:
            y_pred[0][i] = 1
    return y_pred

n_features = X_tr_arr.shape[1]
w, b = initialize_params(n_features)
coeff = model_predict(w, b, X_tr_arr, y_tr_arr, learning_rate=0.0001, no_iterations=4500)

print("coeff['w'] shape:", coeff["w"].shape)
print("Transposing w to (1, 4)...")

w = coeff["w"].T
print("w shape after transpose:", w.shape)

print("X_tr_arr.T shape:", X_tr_arr.T.shape)
final_train_pred = sigmoid_activation(np.dot(w,X_tr_arr.T)+b)
print("final_train_pred shape:", final_train_pred.shape)

m_tr = X_tr_arr.shape[0]
y_tr_pred = predict(final_train_pred, m_tr)
print("y_tr_pred shape:", y_tr_pred.shape)

print("\n All shapes are compatible!")
