# Check for corrupted files
from PIL import Image
import io

def check_images_in_directory(directory):
    corrupted = []
    total = 0
    for class_name in os.listdir(directory):
        class_path = os.path.join(directory, class_name)
        if os.path.isdir(class_path):
            for img_file in os.listdir(class_path):
                total += 1
                img_path = os.path.join(class_path, img_file)
                try:
                    with Image.open(img_path) as img:
                        img.verify()
                except Exception as e:
                    corrupted.append(img_path)
    return total, corrupted

total_train, corrupted_train = check_images_in_directory(train_dir)
total_test, corrupted_test = check_images_in_directory(test_dir)

print(f"\nTrain: {len(corrupted_train)} corrupted out of {total_train}")
print(f"Test: {len(corrupted_test)} corrupted out of {total_test}")
if corrupted_train:
    print("Corrupted train files:", corrupted_train[:3])
