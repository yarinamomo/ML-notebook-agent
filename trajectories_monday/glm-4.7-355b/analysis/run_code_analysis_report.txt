    corrupted = []

    for class_name in os.listdir(directory):
        class_path = os.path.join(directory, class_name)
        if os.path.isdir(class_path):
            for img_file in os.listdir(class_path):

                img_path = os.path.join(class_path, img_file)
                try:
                    with Image.open(img_path) as img:

                except Exception as e:
                    corrupted.append(img_path)
    return total, corrupted

import cv2

def testopencv_loading(directory):
    success = 0
    for class_name in os.listdir(directory)[0:1]:  # Check first class
        class_path = os.path.join(directory, class_name)
        if os.path.isdir(class_path):
            for img_file in os.listdir(class_path)[:5]:  # Check first 5 images
                img_path = os.path.join(class_path, img_file)
                try:
                    img = cv2.imread(img_path)
                    if img is not None:
                        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
                        img = cv2.resize(img, (224, 224))
                        img = img.astype(np.float32) / 255.0
                        success += 1
                        print(f"Loaded with OpenCV: {img_file}, shape: {img.shape}")
                    else:
                        print(f"OpenCV returned None for: {img_file}")
                except Exception as e:
                    print(f"OpenCV failed for {img_file}: {e}")
    return success

print("Training images with OpenCV:")
success_train = testopencv_loading(train_dir)
print(f"\nSuccessfully loaded {success_train} training images with OpenCV")
